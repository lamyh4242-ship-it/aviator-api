<script>
// URL de votre API FastAPI (à modifier après déploiement)
const API_URL = "https://votre-api.onrender.com/analyze"; // ← Remplacez par l'URL réelle

// OCR
const imageInput = document.getElementById('imageInput');
const selectImageBtn = document.getElementById('selectImageBtn');
const ocrBtn = document.getElementById('ocrBtn');
const ocrStatus = document.getElementById('ocrStatus');
const historyInput = document.getElementById('historyInput');
const analyzeBtn = document.getElementById('analyzeBtn');

selectImageBtn.addEventListener('click', ()=> imageInput.click());
imageInput.addEventListener('change', (e)=> {
  if (e.target.files.length) { ocrBtn.disabled = false; ocrStatus.textContent = "Image prête."; }
});
ocrBtn.addEventListener('click', async ()=>{
  let file = imageInput.files[0];
  if (!file) return;
  ocrStatus.textContent = "OCR...";
  ocrBtn.disabled = true;
  try {
    let { data: { text } } = await Tesseract.recognize(file, 'eng+fra');
    let vals = text.match(/\d+[.,]\d+|\d+/g)?.map(s => parseFloat(s.replace(',','.'))).filter(n => n>=1 && n<=50) || [];
    if (vals.length >= 3) {
      historyInput.value = vals.join(',');
      ocrStatus.textContent = `✅ ${vals.length} coeffs.`;
    } else ocrStatus.textContent = "Pas assez de valeurs.";
  } catch(e) { ocrStatus.textContent = "Erreur OCR."; }
  ocrBtn.disabled = false;
});

// Analyse via API
analyzeBtn.addEventListener('click', async () => {
  let raw = historyInput.value.trim();
  let coeffs = raw.match(/\d+[.,]\d+|\d+/g)?.map(s => parseFloat(s.replace(',','.'))).filter(n => n>=1 && n<=50);
  if (!coeffs || coeffs.length < 3) { alert("3 coefficients minimum."); return; }

  let lastTime = document.getElementById('lastTimeInput').value;
  let interval = parseFloat(document.getElementById('intervalInput').value) || 30;

  try {
    let resp = await fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        coefficients: coeffs,
        last_tour_time: lastTime || null,
        interval_seconds: interval,
        future_turns_poisson: 5
      })
    });
    let data = await resp.json();
    if (!resp.ok) throw new Error(data.detail || 'Erreur');
    displayResults(data);
  } catch(e) {
    alert("Erreur API : " + e.message);
  }
});

function displayResults(d) {
  // Stats
  const statsDiv = document.getElementById('statsContainer');
  statsDiv.innerHTML = `
    <div class="stat-item">Volatilité: ${d.volatilite}</div>
    <div class="stat-item">Espérance: ${d.esperance}</div>
    <div class="stat-item">SMA 10: ${d.sma10 ?? '-'}</div>
    <div class="stat-item">SMA 25: ${d.sma25 ?? '-'}</div>
    <div class="stat-item">SMA 50: ${d.sma50 ?? '-'}</div>
    <div class="stat-item">α Pareto: ${d.alpha_pareto}</div>
    <div class="stat-item">Bayes ≥2.5: ${(d.prob_25_bayes*100).toFixed(1)}%</div>
    <div class="stat-item">Bayes ≥5: ${(d.prob_5_bayes*100).toFixed(1)}%</div>
    <div class="stat-item">Markov ≥2.5: ${(d.markov_25*100).toFixed(1)}%</div>
    <div class="stat-item">Poisson ≥2.5: ${(d.poisson_25_5*100).toFixed(1)}%</div>
    <div class="stat-item">LogReg ≥2.5: ${(d.logistic_25*100).toFixed(1)}%</div>
    <div class="stat-item">LogReg ≥5: ${(d.logistic_5*100).toFixed(1)}%</div>
    <div class="stat-item">LogReg ≥10: ${(d.logistic_10*100).toFixed(1)}%</div>
  `;

  // Prédictions d'heures
  const predDiv = document.getElementById('predictionsContainer');
  predDiv.innerHTML = `
    <div class="pred-card">
      <div class="mult">x2.50</div>
      <div>${d.heure_entree_25 || '--'}</div>
      <div class="small">confiance ${(d.logistic_25*100).toFixed(0)}%</div>
    </div>
    <div class="pred-card">
      <div class="mult">x5.00</div>
      <div>${d.heure_entree_5 || '--'}</div>
      <div class="small">confiance ${(d.logistic_5*100).toFixed(0)}%</div>
    </div>
    <div class="pred-card">
      <div class="mult">x10.00</div>
      <div>${d.heure_entree_10 || '--'}</div>
      <div class="small">confiance ${(d.logistic_10*100).toFixed(0)}%</div>
    </div>
  `;

  // Heatmap
  let canvas = document.getElementById('heatmapCanvas');
  let ctx = canvas.getContext('2d');
  let w = canvas.width = canvas.clientWidth;
  let h = 100;
  ctx.clearRect(0,0,w,h);
  let bins = d.heatmap_bins;
  let max = Math.max(...bins,1);
  let colors = ['#2ecc71','#f1c40f','#e67e22','#e74c3c','#8e44ad'];
  let bw = w / bins.length;
  bins.forEach((cnt,i) => {
    let barH = (cnt/max)*h;
    ctx.fillStyle = colors[i];
    ctx.fillRect(i*bw, h-barH, bw-2, barH);
    ctx.fillStyle = 'white';
    ctx.font = '9px monospace';
    ctx.fillText(cnt, i*bw+3, h-barH+12);
  });
}
